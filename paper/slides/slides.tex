\begin{frame}{The throttle principle: an ideal power bound, closed to 4--6\%}
  \begin{block}{The ideal law (measured constants; near-unity escape, $\varphi \ll V$)}
    \centering
    \begin{tabular}{l}
      $F = \sqrt{2m_e/e}\; I\sqrt{\mathrm{KE}}$
        \hfill (beam momentum flux -- $m_e$ and $e$, nothing else)\\[0.4ex]
      $P = V I \;\Rightarrow\; P = F\sqrt{V}/c_{\mathrm{ideal}}$,\quad
        $c_{\mathrm{ideal}} = \sqrt{2m_e/e}$
        \hfill (fixed-thrust bound)\\[0.4ex]
      throttle: \emph{largest feasible escaped current, lowest sufficient
        voltage}\\[0.4ex]
      \footnotesize PIC proves the concept: measured
        $c_F = 0.97\,c_{\mathrm{ideal}}$ (divergence),
        $\kappa = 0.81$ (space charge) $\Rightarrow$ a real geometry runs
        at 1.19--1.22$\times$ the parameter-free bound\\
    \end{tabular}
  \end{block}
  \begin{itemize}\small
    \item Validated against the gated frontier: model 11.7 / 65.8 /
          177.9\,mW vs measured 12.1 / 68.4 / 189\,mW -- within 4--6\%;
          the residual is the charging tax $V/(V-\varphi)$, supplied by the
          measured collection law when a tighter number is wanted.
    \item Feasible current bounded by the emitter, ambient return current,
          spacecraft potential, and beam escape -- acknowledged, not
          optimized here.
    \item Closed-loop optimization for a chosen cathode and geometry:
          future work.
  \end{itemize}
\end{frame}

\begin{frame}{``Why not a tether?''}
  \begin{itemize}\small
    \item Conceded: per watt, electrodynamic-tether thrust at km scale beats
          any reaction engine, including this one.
    \item For the bodies considered here the comparison inverts:
    \begin{itemize}\small
      \item a centimetre-class body or a 3U CubeSat vs.\ a deployed,
            stabilized conductor $10^3$--$10^6\times$ its own length
            (libration control, meteoroid survivability);
      \item thrust constrained to the $I\mathbf{L}\times\mathbf{B}$
            direction;
      \item and a \emph{bare} tether still needs electron collection and
            emission at its ends -- \textbf{the same contactor physics this
            device is, minus the tether}.
    \end{itemize}
    \item Related flown physics is our foundation, not our competitor:
          bare-tether OML collection (Sanmart\'in 1993), PMG plasma
          contactors (1993), E-sail electron gun (Janhunen).
  \end{itemize}
\end{frame}

\begin{frame}{Backup: off-design low-voltage losses are geometry-specific}
  \begin{itemize}\small
    \item Fixed-thrust runs at low voltage measured power up to
          $\approx$1.5--2$\times$ the ideal bound.
    \item The excess decomposes into (i) a non-optimal voltage for the
          demanded thrust and (ii) real losses: beam interception as
          low-voltage optics diverge the plume against the aperture, the
          0.81 energy fraction, emission-type overheads.
    \item The interception is the can's, not the gun's: the identical
          92.4\,V command transmits 99.99\% in an isolated-gun fixture.
    \item The concept therefore carries the ideal bound plus its measured
          closure; loss optimization is engineering future work.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: settle time and the 300\,V late-slope caveat}
  \begin{itemize}\small
    \item Body $\approx$ capacitor charged by the escaping beam:
          $\tau \approx C\varphi/I_{\mathrm{esc}} \approx 28$\,ns at the
          reference; every run spans $\approx$29$\tau$.
    \item At high $\chi$ the sheath equilibrates on the \emph{ion}
          timescale: the 300\,V late slope decayed $+44 \to +27$\,mV/ns but
          was nonzero at 800\,ns.
    \item Settled-value extrapolation: $\varphi_\infty \approx 42$--48\,V --
          inside the 50\,V gate, carried through the $\alpha$ fit as the
          0.85--0.89 band.
    \item The caveat was pre-registered \emph{with} the hypothesis test.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: numerics detail}
  \begin{itemize}\small

\begin{frame}{Backup: slender-body hypotheses --- and the result}
  \begin{itemize}\small
    \item Drag buys the ram silhouette; collection and solar power buy the

          100--200\,mW harvest, tens of grams.
    \item Pre-registered discrimination at the 200\,V anchor demand
          (\O10$\times$30\,mm, 3.24$\times$ area):
          \textbf{A} -- area-only scaling, $\varphi \approx 4$--5\,V;

    \item \textbf{Measured $\varphi = 4.38$\,V} (4.79\,V at run end,

    \item Thin-radius limit connects to flown bare-tether collection
          physics.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: propellant accounting}
  \begin{itemize}\small
    \item ``Propellant'' = ambient electrons: $\dot m = (m_e/e)\,I \approx
          2\times10^{-15}$\,kg/s at 0.342\,mA $\approx$ \textbf{60\,ng/year}.
    \item Effective $I_{\mathrm{sp}} \sim 10^{6}$\,s -- not a merit figure
          (electrons are nearly free momentum-poor mass), quoted only for
          the EP bookkeeping.
    \item Five years of nN thrust $\approx$ 5\,N\,s $\approx$ 0.3\,g of
          electrospray propellant: the argument for this device is never
          propellant \emph{mass} -- it is the \textbf{dry-system floor}
          (tank+feed+valves+PPU $\approx$ 0.3--1\,kg -- a large share of a
          kilogram-class CubeSat before the first newton-second).
  \end{itemize}
\end{frame}
